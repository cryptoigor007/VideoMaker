Behavior:
  «большой риск» (12) → together
  «главный секрет» (14) → split
  «проблема серьёзная» → split

Install (macOS):
  cd VideoMaker_full   # or your project folder
  python3 -m venv .venv
  source .venv/bin/activate
  pip install -r requirements.txt
  pip install mlx-whisper
  brew install ffmpeg
  cp .env.example .env   # add GEMINI_API_KEY
  python -m video_maker.main
  # or: ./start.command

Verify:
  python -m py_compile video_maker/engines/subtitles.py
  python -m pytest tests/test_shorts_parity.py -v
